#include "webserv.hpp"

#define MAX_EVENTS 64
#define EPOLL_TIMEOUT_MS 1000   // wake up every second

void set_event(struct epoll_event *event, int flag, int fd)
{
  event->events = flag;
  event->data.fd = fd;
}

int set_epoll_event(struct epoll_event &event, int fd, int epoll_fd, int new_flag, int change_flag)
{
  set_event(&event, new_flag, fd);
  if (epoll_ctl(epoll_fd, change_flag, fd, &event) == ERROR)
    return print_error("epoll_ctl"), ERROR;
  return SUCCESS;
}

// Assume t_parse_data has a time_t last_activity field.
t_parse_data create_parse_data(Conf& conf_c, Server &server)
{
  t_parse_data client_info;
  client_info.response = "";
  client_info.conf = &conf_c;
  client_info.server = &server;
  client_info.last_activity = time(NULL);   // initialise timestamp
  return client_info;
}

void end_connection(int fd, int epoll_fd, std::map<int, t_parse_data> &client_infos)
{
  print("Connection ceased with fd " + to_str(fd));
  epoll_ctl(epoll_fd, EPOLL_CTL_DEL, fd, NULL);
  close(fd);
  client_infos.erase(fd);
}

// Optional: send a 408 response before closing. For simplicity we just close.
// If you want to send a response, you can call a function similar to send_response()
// but here we just close to avoid complexity.
static void manage_requests(struct epoll_event event,
            int epoll_fd, Conf &conf_c, std::map<int, Server> &servers,
            std::map<int, t_parse_data> &client_infos)
{
  int fd = event.data.fd;

  // 1. Check if fd is a server socket
  std::map<int, Server>::iterator server_it = servers.find(fd);
  if (server_it != servers.end())
  {
    int client_fd = accept_client(fd);
    if (client_fd == ERROR) {
      close(client_fd);
      return;
    }
    set_nonblocking(client_fd);
    struct epoll_event s_event;
    if (set_epoll_event(s_event, client_fd, epoll_fd, EPOLLIN, EPOLL_CTL_ADD) == ERROR)
      return;

    t_parse_data new_client = create_parse_data(conf_c, server_it->second);
    client_infos.insert(std::pair<int, t_parse_data>(client_fd, new_client));
    return;
  }

  // 2. Check if fd exists in client_infos; if not, ignore (stale event)
  std::map<int, t_parse_data>::iterator client_it = client_infos.find(fd);
  if (client_it == client_infos.end())
  {
    // This fd is not a known client; could be a leftover from a closed connection.
    // Just return without doing anything.
    return;
  }
  t_parse_data &info = client_it->second;

  // 3. Handle errors / hang-up
  if (event.events & (EPOLLERR | EPOLLHUP))
  {
    end_connection(fd, epoll_fd, client_infos);
    return;
  }

  // 4. Read event
  if (event.events & EPOLLIN)
  {
    info.last_activity = time(NULL);  // update timestamp on read activity
    if (handle_request(fd, info) == SUCCESS)
    {
      if (set_epoll_event(event, fd, epoll_fd, EPOLLOUT, EPOLL_CTL_MOD) == ERROR)
        return;
    }
    return;
  }

  // 5. Write event (response ready)
  if (event.events & EPOLLOUT)
  {
    info.last_activity = time(NULL);  // update on write activity
    if (send_response(fd, info.response) == UNFINISHED)
    {
      // still more to write, keep EPOLLOUT
      if (set_epoll_event(event, fd, epoll_fd, EPOLLOUT, EPOLL_CTL_MOD) == ERROR)
        return;
    }
    else
    {
      // finished sending, close connection (no keep-alive)
      end_connection(fd, epoll_fd, client_infos);
    }
    return;
  }
}

int manage_events(std::map<int, Server> &servers, Conf &conf_c)
{
  std::map<int, t_parse_data> client_infos;
  int epoll_fd = epoll_create1(0);
  if (epoll_fd == ERROR)
    return ERROR;

  struct epoll_event s_event;
  for (std::map<int, Server>::iterator it = servers.begin(); it != servers.end(); ++it)
  {
    int server_fd = it->first;
    set_event(&s_event, EPOLLIN, server_fd);
    if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, server_fd, &s_event) == ERROR)
    {
      close(epoll_fd);
      return ERROR;
    }
  }

  struct epoll_event events[MAX_EVENTS];
  while (gSignalStatus != 2)
  {
    // Use a finite timeout so we can check for client timeouts periodically.
    int nfds = epoll_wait(epoll_fd, events, MAX_EVENTS, EPOLL_TIMEOUT_MS);
    if (nfds == ERROR)
    {
      if (errno == EINTR) continue;
      perror("epoll_wait");
      return ERROR;
    }

    // -------- Timeout checking (before handling events) --------
   // time_t now = time(NULL);
    std::vector<int> fds_to_close;
    for (std::map<int, t_parse_data>::iterator it = client_infos.begin(); it != client_infos.end(); ++it)
    {
      int fd = it->first;
      if (it->second.request.isTimedOut(fd))
      {
        // optionally send 408 here before closing
        fds_to_close.push_back(fd);
      }
    }

    // Close timed-out connections. Use vector to avoid invalidating iterators during the loop.
    for (std::vector<int>::iterator it = fds_to_close.begin(); it != fds_to_close.end(); ++it)
    {
      int fd = *it;
      // Only close if still in map (may have been closed by an event in the meantime)
      if (client_infos.find(fd) != client_infos.end())
      {
        end_connection(fd, epoll_fd, client_infos);
      }
    }

    // -------- Handle events returned by epoll_wait --------
    for (int i = 0; i < nfds; i++)
    {
      int fd = events[i].data.fd;
      // Skip if this fd is no longer valid (e.g., it was just timed out and closed)
      if (client_infos.find(fd) == client_infos.end() && servers.find(fd) == servers.end())
        continue;

      manage_requests(events[i], epoll_fd, conf_c, servers, client_infos);
    }
  }

  // Cleanup
  for (std::map<int, Server>::iterator it = servers.begin(); it != servers.end(); ++it)
  {
    close(it->first);
  }
  close(epoll_fd);
  return SUCCESS;
}
